### تسجيل حساب جديد

```javascript
const result = app.register('username', 'password123');
if (result.success) {
    console.log('تم إنشاء الحساب بنجاح');
}
```

### تسجيل الدخول

```javascript
const result = app.login('username', 'password123');
if (result.success) {
    console.log('تم تسجيل الدخول بنجاح');
}
```

### بدء جلسة اختبار

```javascript
const session = app.startTestSession(28);
if (session.success) {
    console.log('بدأت جلسة اختبار جديدة');
}
```

### الإجابة على سؤال

```javascript
const result = app.answerQuestion(0, 'ا');
if (result.isCorrect) {
    console.log('إجابة صحيحة! ✅');
} else {
    console.log('إجابة خاطئة ❌');
}
```

### إنهاء الجلسة

```javascript
const results = app.endTestSession();
console.log(`الدقة: ${results.statistics.accuracy}%`);
```
